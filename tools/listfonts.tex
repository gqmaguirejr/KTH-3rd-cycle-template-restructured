\documentclass{article}
\usepackage{luacode}
\usepackage[margin=2cm]{geometry}

\newcommand{\dname}[1]{\textbf{#1}}
\newcommand{\fname}[1]{\texttt{#1}}

\title{List all of the available fonts and populate the font cache}
\author{Gerald Q. Maguire Jr.}
\date{October 2026}

\begin{document}
\maketitle

\begin{luacode}
  maxfonts = 10000

  local exclusion_list = {
    ["HEJI2Text.otf"] = true,
    ["NotoColorEmoji.ttf"] = true,
    ["Cantarell-VF.otf"] = true,
  }

  myfonts = luaotfload.aux.read_font_index()
  local number_of_fonts = #myfonts.mappings

  tex.print("\\section*{Found " .. number_of_fonts .. " available font entries}")
  tex.print("\\clearpage")
  tex.print("\\section*{Font Mapping Basenames}")

  local n = 0
  for _, m in ipairs(myfonts.mappings) do
    n = n + 1
    if n > maxfonts then break end

    local base = m.basename or ""
    local clean_base = string.gsub(base, "_", "\\_")

    -- 1. Check exclusion list or Variable Font pattern
    local is_variable = string.find(base, "%-VF") or string.find(base, "VF%.") or string.find(base, "Variable")

    if exclusion_list[base] or is_variable then
      tex.print("\\ttfamily " .. n .. " " .. clean_base .. " \\textit{[excluded: variable font or unsupported CFF2]}\\par")
    else
      -- 2. Safely define font
      local ok, fid = pcall(function()
        return luaotfload.define_font("file:" .. base, 917504)
      end)

      if ok and type(fid) == "number" and fid > 0 then
        local font_obj = font.getfont(fid)
        local chars = font_obj and font_obj.characters
        local has_ascii = chars and (chars[65] or chars[97] or chars[49])

        if has_ascii then
          tex.print(
            "\\ttfamily " .. n .. " " .. clean_base .. " -- {\\setfontid " .. fid .. " Abc 123}\\par"
          )
        else
          tex.print(
            "\\ttfamily " .. n .. " " .. clean_base .. " \\textit{[non-Latin or symbol; ASCII omitted]}\\par"
          )
        end
      else
        tex.print(
          "\\ttfamily " .. n .. " " .. clean_base .. " \\textit{[unreadable or missing metric data]}\\par"
        )
      end
    end
  end
\end{luacode}

\end{document}
